ZAPYTANIE 1:
SELECT tatuaze.wzor,tatuaze.cena
FROM tatuaze
WHERE tatuaze.kolor = "czarny" and tatuaze.cena > 100
ORDER BY tatuaze.cena desc

ZAPYTANIE 2:
SELECT tatuaze.kolor,count(tatuaze.id)
FROM tatuaze
GROUP BY tatuaze.kolor

ZAPYTANIE 3:
SELECT tatuaze.wzor,style.nazwa
FROM tatuaze
JOIN style ON style.id = tatuaze.style_id
WHERE tatuaze.kolor = "zielony"

ZAPYTANIE 4:
UPDATE tatuaze SET cena = cena * 0.1 WHERE tatuaze.cena = "czerwony"
