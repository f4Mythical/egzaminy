FROM szkolenia
ORDER BY szkolenia.Data

ZAPYTANIE 2:
SELECT szkolenia.Data,szkolenia.Temat,trenerzy.Nazwisko,trenerzy.Imie
FROM szkolenia
JOIN trenerzy ON trenerzy.Id = szkolenia.Id_trenera

ZAPYTANIE 3:
SELECT trenerzy.imie,count(szkolenia.Id_trenera)
FROM trenerzy
JOIN szkolenia ON trenerzy.Id = szkolenia.Id_trenera
GROUP BY trenerzy.id
