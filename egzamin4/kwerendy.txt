SELECT smok.nazwa,smok.dlugosc,smok.szerokosc
FROM smok
WHERE smok.pochodzenie like "polska"
KWERENDA 2:
SELECT DISTINCT smok.pochodzenie
FROM smok
ORDER BY smok.pochodzenie 
KWERENDA 3:
SELECT parada.rok,avg(smok.dlugosc) as 'Średnia długość'
FROM parada
JOIN udzial ON udzial.id_parada = parada.id
JOIN smok ON udzial.id_smok = smok.id
WHERE parada.rok > 2005
GROUP BY parada.rok
