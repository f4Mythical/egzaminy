SELECT zamowienia.plik,zamowienia.nr_klienta,zamowienia.liczba_odbitek
FROM zamowienia
WHERE zamowienia.liczba_odbitek > 100
ZAPYTANIE 2:
SELECT (zamowienia.liczba_odbitek*zdjecia.cena) as "Do zapłaty"
FROM zamowienia
JOIN zdjecia ON zdjecia.rodzaj = zamowienia.rodzaj
WHERE zamowienia.nr_klienta = 3

FROM zamowienia
