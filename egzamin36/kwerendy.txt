ZAPYTANIE 1:
SELECT materialy.nazwa,materialy.ilosc, producenci.nazwaProd
FROM materialy
JOIN producenci ON materialy.Producenci_id = producenci.id
WHERE materialy.ilosc > 9

ZAPYTANIE 2:
SELECT materialy.nazwa,producenci.nazwaProd,jednostki.nazwaJedn
FROM materialy
JOIN jednostki ON jednostki.id = materialy.Jednostki_id
JOIN producenci ON producenci.id = materialy.Producenci_id
